// ----------------------------------------------------------------------------------------------------------------
#include <stdio.h>
#include <stdlib.h>   // malloc、free、NULL 

typedef struct node {
    int data;              // 数据域
    struct node *next;     // 指针域
} Node, *LinkList;         // 默认链表带头结点

// 1. 判断单链表是否递增有序
int IsIncrease(LinkList L)
{
    Node *p = L->next;              // p 指向第一个数据结点
    while (p != NULL && p->next != NULL) {
        if (p->data > p->next->data)  // 前一个比后一个大 → 不是递增
            return 0;
        p = p->next;                // 往后走一步，继续比较下一对
    }
    return 1;                       // 全部比较完都没发现问题 → 递增
}

// 2. 删除循环单链表中 s 的前趋结点
void DeletePre(Node *s)
{
    Node *pre = s;                  // 从 s 出发，沿环找 s 前驱的前驱
    while (pre->next->next != s)    // 停下时：pre->next->next == s
        pre = pre->next;
    Node *q = pre->next;            // q 就是 s 的前趋结点
    pre->next = s;                  // pre 直接跨过 q 牵住 s
    free(q);                        // 释放 q，删除完成
}

// 3. 将线性表就地扩展 2 倍：(a₁,a₂,…,aₙ) → (a₁,a₁,a₂,a₂,…,aₙ,aₙ)
void Expand(LinkList L)
{
    Node *p = L->next, *q;          // p 指向当前处理的“原件”结点
    while (p != NULL) {
        q = (Node *)malloc(sizeof(Node));  // 造一个新结点
        q->data = p->data;          // 新结点复制原件的数据
        q->next = p->next;          // 新结点接手 p 原来的后继
        p->next = q;                // 再让 p 指向新结点 → 完成插入
        p = q->next;                // 跳过副本，p 落到下一个“原件”上
    }
}

// ============ 以下是自测代码 ============

// 工具：尾插法建一条带头结点的单链表，数据取自数组 a[0..n-1]
LinkList CreateList(int a[], int n)
{
    LinkList head = (Node *)malloc(sizeof(Node));  // 头结点不存数据
    Node *tail = head;                             // tail 永远指向当前尾结点
    for (int i = 0; i < n; i++) {
        Node *s = (Node *)malloc(sizeof(Node));
        s->data = a[i];
        tail->next = s;                            // 新结点接到尾部
        tail = s;                                  // tail 后移到新的尾
    }
    tail->next = NULL;                             // 尾结点置空，链表收尾
    return head;
}

// 工具：打印带头结点的单链表
void PrintList(LinkList L)
{
    for (Node *p = L->next; p != NULL; p = p->next)
        printf("%d ", p->data);
    printf("\n");
}

int main(void)
{
    // ---- 测试 1：IsIncrease，递增返回 1，否则返回 0 ----
    int a1[] = {1, 3, 5, 7, 9};
    int a2[] = {1, 3, 2, 7, 9};            // 3 后面跟了 2，破坏递增
    LinkList L1 = CreateList(a1, 5);
    LinkList L2 = CreateList(a2, 5);
    printf("[IsIncrease] {1,3,5,7,9} -> %d（期望 1）\n", IsIncrease(L1));
    printf("[IsIncrease] {1,3,2,7,9} -> %d（期望 0）\n", IsIncrease(L2));

    // ---- 测试 2：Expand，每个结点后面插一个相同数据的副本 ----
    printf("[Expand  前] ");
    PrintList(L1);
    Expand(L1);
    printf("[Expand  后] ");
    PrintList(L1);                         // 期望：1 1 3 3 5 5 7 7 9 9

    // ---- 测试 3：DeletePre，注意它处理的是“循环”单链表 ----
    // 手工搭一条循环链表：头结点 → 10 → 20 → 30 → 绕回头结点
    Node *head = (Node *)malloc(sizeof(Node));
    Node *s1   = (Node *)malloc(sizeof(Node));
    Node *s2   = (Node *)malloc(sizeof(Node));
    Node *s    = (Node *)malloc(sizeof(Node));
    head->data = 0;                        // 头结点数据不使用
    s1->data = 10;
    s2->data = 20;
    s->data  = 30;
    head->next = s1;
    s1->next = s2;
    s2->next = s;
    s->next = head;                        // 尾巴绕回头结点，成环！
    DeletePre(s);                          // 删掉 30 的前驱 20
    printf("[DeletePre]  10 20 30 删 30 的前驱 -> ");
    for (Node *p = head->next; p != head; p = p->next)   // 绕一圈打印
        printf("%d ", p->data);            // 期望：10 30
    printf("\n");

    return 0;
}

## 先铺垫：三道题共用的结构体（第 1–6 行）
```plaintext
typedef struct node {
    int data;              // 数据域
    struct node *next;     // 指针域
} Node, *LinkList;
```

- `struct node` ：链表的 一个结点 ，像一节车厢，装两样东西
  - `int data` ：车厢里的 货 （一个整数）
  - `struct node *next` ：这节车厢 勾着下一节的钩子
- `typedef` ：给复杂类型起别名
  - `Node` =`struct node` ，以后写`Node` 就代表"一个结点"
  - `LinkList` =`Node *` ，也就是"指向结点的指针"，通常拿它指向 头结点 （带头结点的单链表）
- 带头结点：链表第一个结点不存有效数据，只是当个"门面"，门面后面才是真正装货的结点。好处是空表和非空表的处理逻辑统一
画出来长这样 （带头结点的链表，存 {10, 20, 30}）：
```plaintext
[头结点(data无用)] → [10] → [20] → [30] → NULL
     ↑ LinkList L  ↑ L->next  ↑       ↑
```

## 第 1 题：IsIncrease —— 判断单链表是否递增有序
### 题目意思
给你一条单链表，从头到尾，每个结点的值是不是都比后一个小或相等？比如`{1, 3, 5, 7}` 递增 → 返回 1；`{1, 3, 2, 7}` 里 3 比 2 大 → 返回 0。

### 完整代码（第 8–17 行）
```plaintext
int IsIncrease(LinkList L)
{
    Node *p = L->next;              // 第 10 行
    while (p != NULL && p->next != NULL) {   // 第 11 行
        if (p->data > p->next->data)        // 第 12 行
            return 0;                        // 第 13 行
        p = p->next;                        // 第 14 行
    }
    return 1;                               // 第 16 行
}
```

### 逐行拆
第 9 行`int IsIncrease(LinkList L)`

- 函数名 IsIncrease = "是否递增"
- 参数`LinkList L` ：传进来一条带头结点的链表（就是那个指向头结点的指针）
- 返回值`int` ：C 语言里常用 1 表示"是"、0 表示"否"


第 10 行`Node *p = L->next;`

- `p` 是个指针变量，用来 指着当前正在看的结点
- `L->next` 是什么？`L` 指着头结点，头结点的`next` 指向 第一个真正装数据的结点
- 所以这行做完，`p` 就站在第一个数据结点上了
- 为什么不直接`p = L` ？因为头结点不存有效数据，从它开始比没意义


第 11 行`while (p != NULL && p->next != NULL) {`

- 这是循环，只要两个条件都成立就继续比
- 条件 1`p != NULL` ：当前结点还存在（没走到链表末尾）
- 条件 2`p->next != NULL` ： 下一个结点也存在 （不然没的比）
- 两个条件用`&&` 连：意思是"我手上这个结点和它后面那个都在，才能做比较"
- 举例：链表只有一个结点`{5}` ，那么`p = 指向5` ，`p->next = NULL` ，第二个条件为假，循环一次都不进，直接走到第 16 行返回 1（单结点天然是递增的，没问题）


第 12 行`if (p->data > p->next->data)`

- `p->data` ：当前结点里的货
- `p->next->data` ：下一个结点里的货
- `>` 不是`>=` ！严格来说递增应该是`<=` （允许相等），但这里用`>` ，意思是"如果前一个 大于 后一个"，一旦发现破坏递增，立刻
- 举例：当前是 3，下一个是 2，3 > 2 成立 → 递增被打破


第 13 行`return 0;`

- 发现反例，直接返回 0， 函数到此结束
- 这叫"提前返回"，效率高，不用遍历完


第 14 行`p = p->next;`

- 没发现反例，p 往后挪一步，去看下一对
- 图示：p 从指着 A 变成指着 B，准备比较 B 和 C


第 16 行`return 1;`

- 循环结束了，说明 从头到尾没有任何一对是前大后小
- 整条链表递增有序，返回 1

## 第 2 题：DeletePre —— 删除循环单链表中 s 的前驱结点
### 先讲什么是循环单链表
普通单链表的最后一个结点`next = NULL` （断掉）；循环单链表的最后一个结点`next` 指回头结点 ，首尾相连成一个环，没有 NULL 了。
```plaintext
普通：[头] → [10] → [20] → [30] → NULL
循环：[头] → [10] → [20] → [30] → [头]  ← 30 的 next 指回头结点
```

### 题目意思
给你循环链表中的一个结点`s` ，删掉它 前面那个结点 （前驱）。比如链表是`头→10→20→30→头` ，如果`s` 指向 30，那它的前驱是 20，删掉后变成`头→10→30→头` 。

### 完整代码（第 19–28 行）
```plaintext
void DeletePre(Node *s)
{
    Node *pre = s;                  // 第 21 行
    while (pre->next->next != s)    // 第 22 行
        pre = pre->next;            // 第 23 行
    Node *q = pre->next;            // 第 24 行
    pre->next = s;                  // 第 25 行
    free(q);                        // 第 26 行
}
```

### 逐行拆
第 20 行`void DeletePre(Node *s)`

- 函数不返回值（void），只做事
- 参数`Node *s` ：传进来一个结点指针，从这个结点出发找它的前驱


第 21 行`Node *pre = s;`

- `pre` 是个"探路指针"，从`s` 本人开始出发
- 为什么从 s 开始而不是从头结点开始？因为是循环链表，从任何一个点出发都能绕一圈，从 s 出发最省事


第 22 行`while (pre->next->next != s)`

- 这是整道题的核心！条件是"pre 后面第二个结点还不是 s"，就继续走
- 关键推理：
  - 假设`pre` 指着 A，`pre->next` 指着 B，`pre->next->next` 指着 C
  - 如果 C 不是 s → 说明 A 还没走到"s 的前驱的前驱"，继续走
  - 当循环结束时，`pre->next->next == s` ，也就是：
    - `pre` 指着 s 的 前驱的前驱
    - `pre->next` 指着 s 的 前驱 （这就是要删的！）
    - `pre->next->next` 就是 s 自己

- 图示推导（s 指向 30）：
  ```plaintext
初始：pre = s → 30
  循环第1次：pre->next->next = 头->10 = 10 ≠ 30 → 走
  pre = 30 → pre->next->next = 头->10 = 10 ≠ 30 → 走
  等等不对...让我重新推：
  链表：头 → 10 → 20 → 30 → 头
  s = 30
  
  初始：pre = 30
  pre->next->next = 头->10 = 10 ≠ 30 → pre = 头
  pre->next->next = 10->20 = 20 ≠ 30 → pre = 10
  pre->next->next = 20->30 = 30 == 30 → 停！
  
  停在 pre = 10
  pre->next = 20（这就是 s=30 的前驱，要删的！）✓
  ```


第 23 行`pre = pre->next;`

- 循环体就这一句，pre 绕着环往后挪一步


第 24 行`Node *q = pre->next;`

- `q` 用来接住"要删掉的那个结点"（也就是 s 的前驱）
- 为什么要接？因为第 25 行`pre->next = s` 会把 pre 和 q 的连线剪断，不提前存住就找不到 q 了，没法 free


第 25 行`pre->next = s;`

- 把 pre 的钩子从 q 身上摘下来，直接勾到 s 身上
- 这一步 q 就从链表中摘除了，但内存还没释放，所以需要第 26 行


第 26 行`free(q);`

- 把 q 占用的内存还给操作系统
- 这是 C 语言手动管理内存的必须步骤，不 free 就内存泄漏

## 第 3 题：Expand —— 把链表扩展成两倍长
### 题目意思
把链表每个结点后面插一个 和自己数据相同 的副本。比如`{1, 3, 5}` 变成`{1, 1, 3, 3, 5, 5}` 。

### 完整代码（第 31–40 行）
```plaintext
void Expand(LinkList L)
{
    Node *p = L->next, *q;                  // 第 33 行
    while (p != NULL) {                     // 第 34 行
        q = (Node *)malloc(sizeof(Node));   // 第 35 行
        q->data = p->data;                 // 第 36 行
        q->next = p->next;                 // 第 37 行
        p->next = q;                       // 第 38 行
        p = q->next;                       // 第 39 行
    }
}
```

### 逐行拆
第 32 行`void Expand(LinkList L)`

- 不返回值，直接改原链表（原地扩展）
- 参数`LinkList L` ：带头结点的链表


第 33 行`Node *p = L->next, *q;`

- 同时声明两个指针
- `p` ：探路指针，从第一个数据结点开始，逐一看过每一个"原件"
- `q` ：临时指针，用来接住新建出来的"副本"


第 34 行`while (p != NULL)`

- 普通链表的末尾是 NULL，走到 NULL 就停
- 为什么不用像第 1 题那样判断`p->next != NULL` ？因为这里即使 p 是最后一个结点（p->next = NULL），也要给它造一个副本插进去，不能跳过


第 35 行`q = (Node *)malloc(sizeof(Node));`

- `malloc` ：向操作系统申请一块内存，大小刚好够装一个`Node`
- `sizeof(Node)` ：一个结点占多少字节（int 4 字节 + 指针 8 字节 = 12 字节左右）
- `(Node *)` ：malloc 返回的是"无类型指针"，要强转成`Node *` 才能当结点用
- 这行做完，`q` 就指着一块 崭新的、空的结点内存 ，等着往里填数据


第 36 行`q->data = p->data;`

- 把原件的数据 复制 给副本
- p 是原件，q 是副本，这行 = "货一样"


第 37 行`q->next = p->next;`

- 把副本的钩子 先勾上原件原来的下一个结点
- 必须这行先做！顺序很重要！

- 图示（原件 p 后面原来连着 X）：
  ```plaintext
操作前：  p → X
  q->next = p->next 之后：  q → X（同时 p 还 → X）
  ```


第 38 行`p->next = q;`

- 把原件的钩子 改勾上副本
- 这行之后：
  ```plaintext
操作后：  p → q → X
  链就接好了：原件→副本→原件原来的后继
  ```


第 39 行`p = q->next;`

- 关键！ 这行决定了 p 下一站去哪
- 副本 q 插好了，p 不能只走一步到 q（那下一轮又会给 q 造副本，无限循环）
- p 必须 跳过副本 ，直接落到 q 的下一个（也就是原件原来的后继），去处理下一个原件
- 图示：
  ```plaintext
当前：... → [p原件] → [q副本] → [下一个原件] → ...
                          ↑
              p = q->next 就是跳到这里
  ```
为什么 37、38 行顺序不能反？

如果先写`p->next = q` ，那原件的后继就断了，`p->next` 现在是 q，再写`q->next = p->next` 等于`q->next = q` ，副本变成自己勾自己，链表就乱了。 先接后面，再接前面 ，这是链表插入的黄金顺序。

## 三道题的核心套路总结
套路 对应题 一句话 遍历链表比相邻对 IsIncrease 站在当前，往后瞟一眼，不符合就走 双指针沿环走 DeletePre pre 从 s 出发，走到 pre->next->next == s 停 原地插副本 Expand 先接后面再接前面，p 跳一步避开副本
